-- ============================================================ -- English Elfishawy — Security & Feature Setup -- Run this ONCE in Supabase: Dashboard > SQL Editor > New query > Run -- ============================================================ -- 1) Make sure lesson_codes can be tied to one specific student alter table lesson_codes add column if not exists student_id uuid references students(id); -- Make sure one student can only have one access row per lesson create unique index if not exists lesson_access_student_lesson_key on lesson_access(student_id, lesson_id); -- ============================================================ -- 2) Admin check helper -- Change the email below if you ever use a different admin account -- ============================================================ drop function if exists is_admin() cascade; create function is_admin() returns boolean language sql stable as $$ select (auth.jwt() ->> 'email') = 'maryammostafa208125@gmail.com'; $$; -- ============================================================ -- 3) Lock down every table with Row Level Security -- ============================================================ alter table students enable row level security; alter table lessons enable row level security; alter table lesson_access enable row level security; alter table lesson_codes enable row level security; alter table platform_store enable row level security; drop policy if exists "students select own or admin" on students; create policy "students select own or admin" on students for select using (auth.uid() = id or is_admin()); drop policy if exists "students insert own" on students; create policy "students insert own" on students for insert with check (auth.uid() = id); drop policy if exists "students update own or admin" on students; create policy "students update own or admin" on students for update using (auth.uid() = id or is_admin()); drop policy if exists "lessons admin write" on lessons; create policy "lessons admin write" on lessons for all using (is_admin()) with check (is_admin()); drop policy if exists "access select own or admin" on lesson_access; create policy "access select own or admin" on lesson_access for select using (auth.uid() = student_id or is_admin()); drop policy if exists "access admin write" on lesson_access; create policy "access admin write" on lesson_access for all using (is_admin()) with check (is_admin()); -- Note: students never get a direct write policy here on purpose. -- They can only unlock a lesson through redeem_lesson_code() below. drop policy if exists "codes admin only" on lesson_codes; create policy "codes admin only" on lesson_codes for all using (is_admin()) with check (is_admin()); -- Students have NO policy on this table at all, so they can never -- read or guess codes directly — only through the function below. drop policy if exists "settings public read" on platform_store; create policy "settings public read" on platform_store for select using (true); drop policy if exists "settings admin write" on platform_store; create policy "settings admin write" on platform_store for all using (is_admin()) with check (is_admin()); -- ============================================================ -- 4) Safe public view of lessons (title/description only — -- NEVER the video, so a locked lesson's video can't be read -- from the browser before it's unlocked) -- ============================================================ drop view if exists public_lessons; create view public_lessons as select id, title, description, created_at from lessons; grant select on public_lessons to authenticated, anon; -- ============================================================ -- 5) Redeem a code — the ONLY way a student can unlock a lesson. -- Checks the code belongs to THIS student and THIS lesson, -- and hasn't been used before. -- ============================================================ drop function if exists redeem_lesson_code(text, uuid); create function redeem_lesson_code(p_code text, p_lesson_id uuid) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_row lesson_codes%rowtype; begin select * into v_row from lesson_codes where code = upper(trim(p_code)) and lesson_id = p_lesson_id and used = false and student_id = auth.uid() limit 1; if not found then return jsonb_build_object('success', false, 'message', 'الكود غير صحيح أو مستخدم أو مش لحسابك'); end if; update lesson_codes set used = true, used_by = auth.uid() where id = v_row.id and used = false; insert into lesson_access (student_id, lesson_id, status) values (auth.uid(), p_lesson_id, 'unlocked') on conflict (student_id, lesson_id) do update set status = 'unlocked'; return jsonb_build_object('success', true); end; $$; grant execute on function redeem_lesson_code(text, uuid) to authenticated; -- ============================================================ -- 6) Get the video URL — only returned if the student actually -- has that lesson unlocked (or the caller is the admin) -- ============================================================ drop function if exists get_lesson_video(uuid); create function get_lesson_video(p_lesson_id uuid) returns text language plpgsql security definer set search_path = public as $$ declare v_status text; v_url text; begin if is_admin() then select video_url into v_url from lessons where id = p_lesson_id; return v_url; end if; select status into v_status from lesson_access where student_id = auth.uid() and lesson_id = p_lesson_id; if v_status in ('unlocked', 'completed') then select video_url into v_url from lessons where id = p_lesson_id; return v_url; end if; return null; end; $$; grant execute on function get_lesson_video(uuid) to authenticated; -- ============================================================ -- 7) Storage bucket for uploaded lesson videos -- ============================================================ insert into storage.buckets (id, name, public) values ('lesson-videos', 'lesson-videos', true) on conflict (id) do nothing; drop policy if exists "public read lesson videos" on storage.objects; create policy "public read lesson videos" on storage.objects for select using (bucket_id = 'lesson-videos'); drop policy if exists "admin upload lesson videos" on storage.objects; create policy "admin upload lesson videos" on storage.objects for insert with check (bucket_id = 'lesson-videos' and is_admin()); drop policy if exists "admin delete lesson videos" on storage.objects; create policy "admin delete lesson videos" on storage.objects for delete using (bucket_id = 'lesson-videos' and is_admin()); -- ============================================================ -- Done.
